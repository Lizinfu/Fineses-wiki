.RelPermalink -}} {{- else -}} {{- $path := strings.TrimSpace $configured -}} {{- if $path -}} {{- $unsafe := or (gt (len (findRE `(?i)^[a-z][a-z0-9+.-]*:` $path)) 0) (strings.HasPrefix $path "//") -}} {{- $resource := $page.Resources.Get $resourcePath -}} {{- if not $resource -}} {{- with $page.Parent -}} {{- $resource = .Resources.Get $resourcePath -}} {{- end -}} {{- end -}} {{- if not "style" $style -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
